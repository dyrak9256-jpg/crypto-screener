package app

import (
	"context"
	"errors"
	"fmt"
	"log"
	"runtime"
	"strings"
	"sync"
	"sync/atomic"
	"time"

	"crypto-screener/internal/adapters/binance"
	"crypto-screener/internal/domain"

	"github.com/shopspring/decimal"
)

const (
	userPersistenceTimeout = 5 * time.Second
)

var ErrApplicationAlreadyStarted = errors.New("application has already been started")

type Application struct {
	connManager *ConnectorManager
	aggregator  *ShardedAggregator
	tracker     *Tracker
	fundingMgr  *FundingManager
	config      *domain.ScreenerConfig

	adminMu  sync.RWMutex
	adminIDs []int64

	userMgr  *domain.UserManager
	userRepo domain.UserRepository
	router   *NotificationRouter

	tickChan    chan domain.MarketTick
	trackerChan chan domain.SpreadEvent
	dbChan      chan *domain.ArbitrageSignal

	// Текущий контекст приложения.
	ctx atomic.Pointer[context.Context]

	// Application является одноразовым объектом.
	// После закрытия каналов безопасный повторный Run невозможен.
	started atomic.Bool

	// При false новые команды, изменяющие состояние, не принимаются.
	accepting atomic.Bool

	// Отдельные WaitGroup нужны для строгого порядка drain:
	//
	// connector producers
	//       ↓
	// tickChan
	//       ↓
	// ingestionWg
	//       ↓
	// trackerChan
	//       ↓
	// trackerWg
	//       ↓
	// dbChan
	//       ↓
	// persistWg
	//
	ingestionWg sync.WaitGroup
	trackerWg   sync.WaitGroup
	persistWg   sync.WaitGroup

	// DB операции пользователей (/start, /stop и т.д.).
	userWg sync.WaitGroup
}

func NewApplication(
	cfg *domain.ScreenerConfig,
	repo domain.SignalRepository,
	userRepo domain.UserRepository,
) *Application {
	tickChan := make(chan domain.MarketTick, 200_000)
	trackerChan := make(chan domain.SpreadEvent, 50_000)
	dbChan := make(chan *domain.ArbitrageSignal, 10_000)

	userMgr := domain.NewUserManager()
	fundingMgr := NewFundingManager(cfg)

	aggregator := NewShardedAggregator(
		trackerChan,
		fundingMgr,
		cfg,
	)

	router := NewNotificationRouter(
		userMgr,
		aggregator,
		nil,
	)

	tracker := NewTracker(
		cfg,
		dbChan,
		router,
	)

	connMgr := NewConnectorManager(
		tickChan,
		fundingMgr,
	)

	return &Application{
		connManager: connMgr,
		aggregator:  aggregator,
		tracker:     tracker,
		fundingMgr:  fundingMgr,
		config:      cfg,
		userMgr:     userMgr,
		userRepo:    userRepo,
		router:      router,
		tickChan:    tickChan,
		trackerChan: trackerChan,
		dbChan:      dbChan,
	}
}

func (a *Application) SetTelegramSender(tg domain.TelegramSender) {
	a.router.telegram = tg
}

func (a *Application) SetAdminIDs(ids []int64) {
	// Копируем slice, чтобы внешний код не мог изменить его
	// во время чтения Application.
	copied := append([]int64(nil), ids...)

	a.adminMu.Lock()
	a.adminIDs = copied
	a.adminMu.Unlock()
}

func (a *Application) GetConnectorManager() *ConnectorManager {
	return a.connManager
}

func (a *Application) isAdmin(chatID int64) bool {
	a.adminMu.RLock()
	defer a.adminMu.RUnlock()

	if len(a.adminIDs) == 0 {
		return true
	}

	for _, id := range a.adminIDs {
		if chatID == id {
			return true
		}
	}

	return false
}

// getContext возвращает контекст приложения без блокировки.
func (a *Application) getContext() context.Context {
	if ptr := a.ctx.Load(); ptr != nil {
		return *ptr
	}

	return context.Background()
}

func (a *Application) Run(
	ctx context.Context,
	repo domain.SignalRepository,
) error {
	if !a.started.CompareAndSwap(false, true) {
		return ErrApplicationAlreadyStarted
	}

	if ctx == nil {
		ctx = context.Background()
	}

	a.ctx.Store(&ctx)

	log.Println("🚀 Starting High-Performance Arbitrage Engine...")

	// Загружаем пользователей из БД в кэш.
	users, err := a.userRepo.GetAllUsers(ctx)
	if err != nil {
		log.Printf("⚠️ Failed to load users from DB: %v", err)
	} else {
		for _, u := range users {
			a.userMgr.SetUser(u)
		}

		log.Printf("✅ Loaded %d users from DB", len(users))
	}

	// PersistenceWorker стартует первым.
	a.persistWg.Add(1)

	go NewPersistenceWorker(
		a.dbChan,
		repo,
	).Start(ctx, &a.persistWg)

	// Ingestion workers.
	workerCount := runtime.NumCPU() * 2
	if workerCount < 8 {
		workerCount = 8
	}

	for i := 0; i < workerCount; i++ {
		a.ingestionWg.Add(1)
		go a.ingestionWorker()
	}

	// Tracker workers.
	for i := 0; i < 4; i++ {
		a.trackerWg.Add(1)
		go a.trackerWorker()
	}

	a.accepting.Store(true)

	log.Printf(
		"✅ Engine started: %d ingestion workers, 4 tracker workers",
		workerCount,
	)

	<-ctx.Done()

	a.accepting.Store(false)

	log.Println("🛑 Graceful shutdown initiated...")

	// ============================================================
	// SHUTDOWN PIPELINE
	//
	// 1. Остановить всех producers.
	// 2. Закрыть tickChan.
	// 3. Дождаться ingestion workers.
	// 4. Закрыть trackerChan.
	// 5. Дождаться tracker workers.
	// 6. Закрыть dbChan.
	// 7. Дождаться persistence.
	// 8. Дождаться notification goroutines.
	// 9. Дождаться пользовательских DB операций.
	// ============================================================

	// ------------------------------------------------------------
	// [1/9] ConnectorManager
	//
	// После StopAll() ни один connector больше не должен писать
	// в tickChan.
	// ------------------------------------------------------------
	a.connManager.StopAll()

	log.Println("   ↳ [1/9] all connector producers stopped")

	// ------------------------------------------------------------
	// [2/9] Закрываем tickChan.
	// ------------------------------------------------------------
	close(a.tickChan)

	log.Println("   ↳ [2/9] tickChan closed, draining ingestion queue...")

	// ------------------------------------------------------------
	// [3/9] Ingestion workers drain tickChan.
	// ------------------------------------------------------------
	a.ingestionWg.Wait()

	log.Println("   ↳ [3/9] ingestion workers stopped")

	// ------------------------------------------------------------
	// [4/9] Теперь ProcessTick() больше нигде не выполняется,
	// следовательно никто больше не пишет в trackerChan.
	// ------------------------------------------------------------
	close(a.trackerChan)

	log.Println("   ↳ [4/9] trackerChan closed, draining tracker queue...")

	// ------------------------------------------------------------
	// [5/9] Tracker workers drain trackerChan.
	// ------------------------------------------------------------
	a.trackerWg.Wait()

	log.Println("   ↳ [5/9] tracker workers stopped")

	// ------------------------------------------------------------
	// [6/9] Теперь Tracker больше не пишет в dbChan.
	// ------------------------------------------------------------
	close(a.dbChan)

	log.Println("   ↳ [6/9] dbChan closed, draining persistence queue...")

	// ------------------------------------------------------------
	// [7/9] PersistenceWorker сохраняет всё, что осталось в dbChan.
	// ------------------------------------------------------------
	a.persistWg.Wait()

	log.Println("   ↳ [7/9] persistence worker stopped")

	// ------------------------------------------------------------
	// [8/9] Ждём уже запущенные notification goroutines.
	// ------------------------------------------------------------
	if err := a.tracker.Stop(5 * time.Second); err != nil {
		log.Printf("⚠️ Tracker notification shutdown warning: %v", err)
	}

	log.Println("   ↳ [8/9] tracker notifications stopped")

	// ------------------------------------------------------------
	// [9/9] Ждём DB операции пользователей.
	// ------------------------------------------------------------
	a.userWg.Wait()

	log.Println("   ↳ [9/9] user persistence operations stopped")

	log.Println("✅ Shutdown complete.")

	return nil
}

func (a *Application) ingestionWorker() {
	defer a.ingestionWg.Done()

	// После закрытия tickChan worker должен drain'ить уже принятые
	// market ticks, а не немедленно выходить из-за ctx.Done().
	for tick := range a.tickChan {
		a.aggregator.ProcessTick(tick)
	}
}

func (a *Application) trackerWorker() {
	defer a.trackerWg.Done()

	// Аналогично: trackerChan закрывается только после того,
	// как ingestion workers закончили писать в него.
	for event := range a.trackerChan {
		a.tracker.HandleEvent(event)
	}
}

func (a *Application) persistUser(
	user *domain.User,
) {
	if user == nil {
		return
	}

	a.userWg.Add(1)

	go func(u domain.User) {
		defer a.userWg.Done()

		// Сохранение пользовательского состояния не должно
		// автоматически отменяться вместе с Application ctx.
		ctx, cancel := context.WithTimeout(
			context.WithoutCancel(a.getContext()),
			userPersistenceTimeout,
		)
		defer cancel()

		if err := a.userRepo.SaveUser(ctx, &u); err != nil {
			log.Printf("⚠️ SaveUser %d: %v", u.ChatID, err)
		}
	}(*user)
}

func (a *Application) deleteUser(
	chatID int64,
) {
	a.userWg.Add(1)

	go func(cid int64) {
		defer a.userWg.Done()

		ctx, cancel := context.WithTimeout(
			context.WithoutCancel(a.getContext()),
			userPersistenceTimeout,
		)
		defer cancel()

		if err := a.userRepo.DeleteUser(ctx, &cid); err != nil {
			log.Printf("⚠️ DeleteUser %d: %v", cid, err)
		}
	}(chatID)
}

func (a *Application) HandleCommand(
	chatID int64,
	username string,
	cmd string,
	args []string,
) string {
	if !a.accepting.Load() {
		return "⚠️ Двигатель сейчас остановлен."
	}

	appCtx := a.getContext()

	// Проверка прав администратора.
	adminOnly := map[string]bool{
		"addex": true,
		"rmex":  true,
	}

	if adminOnly[cmd] && !a.isAdmin(chatID) {
		return "⛔ Access Denied."
	}

	user, exists := a.userMgr.GetUser(chatID)

	if !exists && cmd != "start" {
		return "⚠️ Вы не подписаны. Отправьте /start для начала работы."
	}

	switch cmd {
	case "start":
		if exists {
			return "✅ Вы уже подписаны! /help — список команд."
		}

		newUser := &domain.User{
			ChatID:    chatID,
			Username:  username,
			MinSpread: a.config.GetHardMinSpread(),
			MinVolume: a.config.GetHardMinVolume(),
			Timeframe: domain.TF_15m,
		}

		a.userMgr.SetUser(newUser)
		a.persistUser(newUser)

		return fmt.Sprintf(
			"🎉 Добро пожаловать, @%s!\nВы подписаны на сигналы.\n/help — список команд.",
			username,
		)

	case "stop":
		a.userMgr.RemoveUser(chatID)
		a.deleteUser(chatID)

		return "👋 Вы отписались. /start — чтобы вернуться."

	case "setcross":
		if len(args) < 1 {
			return "Usage: /setcross <percent>"
		}

		val, err := decimal.NewFromString(args[0])
		if err != nil || val.IsNegative() {
			return "❌ Некорректное число. Пример: /setcross 1.5"
		}

		// Нельзя задавать бессмысленно огромное значение.
		if val.GreaterThan(decimal.NewFromInt(100)) {
			return "❌ Спред не может быть больше 100%."
		}

		spread := val.Div(decimal.NewFromInt(100))

		if spread.LessThan(a.config.GetHardMinSpread()) {
			spread = a.config.GetHardMinSpread()
		}

		updated := *user
		updated.MinSpread = spread

		a.userMgr.SetUser(&updated)
		a.persistUser(&updated)

		return fmt.Sprintf(
			"✅ Минимальный спред: %s%%",
			spread.Mul(decimal.NewFromInt(100)).StringFixed(2),
		)

	case "setvol":
		if len(args) < 1 {
			return "Usage: /setvol <usdt_amount>"
		}

		vol, err := decimal.NewFromString(args[0])
		if err != nil || vol.IsNegative() {
			return "❌ Некорректный объём. Пример: /setvol 500000"
		}

		// Защитный верхний предел.
		if vol.GreaterThan(decimal.NewFromInt(1_000_000_000)) {
			return "❌ Объём слишком большой."
		}

		if vol.LessThan(a.config.GetHardMinVolume()) {
			vol = a.config.GetHardMinVolume()
		}

		updated := *user
		updated.MinVolume = vol

		a.userMgr.SetUser(&updated)
		a.persistUser(&updated)

		return fmt.Sprintf(
			"✅ Минимальный объём: $%s",
			vol.StringFixed(0),
		)

	case "settimeframe":
		if len(args) < 1 {
			return "Usage: /settimeframe <1m|5m|15m|30m|1h|4h|24h>"
		}

		tf := domain.Timeframe(strings.ToLower(args[0]))

		valid := map[domain.Timeframe]bool{
			domain.TF_1m:  true,
			domain.TF_5m:  true,
			domain.TF_15m: true,
			domain.TF_30m: true,
			domain.TF_1h:  true,
			domain.TF_4h:  true,
			domain.TF_24h: true,
		}

		if !valid[tf] {
			return "❌ Неверный таймфрейм. Доступны: 1m, 5m, 15m, 30m, 1h, 4h, 24h"
		}

		updated := *user
		updated.Timeframe = tf

		a.userMgr.SetUser(&updated)
		a.persistUser(&updated)

		return fmt.Sprintf(
			"✅ Таймфрейм: %s",
			tf,
		)

	case "help":
		return "📋 *Доступные команды:*\n" +
			"/start — Подписаться на сигналы\n" +
			"/stop — Отписаться\n" +
			"/setcross <%> — Мин. спред (пример: 1.5)\n" +
			"/setvol <USDT> — Мин. объём (пример: 500000)\n" +
			"/settimeframe <tf> — Таймфрейм (1m, 5m, 15m, 30m, 1h, 4h, 24h)"

	case "addex":
		if len(args) < 1 {
			return "Usage: /addex <exchange>"
		}

		name := strings.ToUpper(args[0])

		var conn domain.ExchangeConnector

		switch name {
		case "BINANCE":
			conn = binance.NewAdapter()

		default:
			return fmt.Sprintf(
				"❌ Exchange %s not supported",
				name,
			)
		}

		if err := a.connManager.AddConnector(
			appCtx,
			name,
			conn,
		); err != nil {
			return fmt.Sprintf(
				"❌ Failed to add %s: %v",
				name,
				err,
			)
		}

		return fmt.Sprintf(
			"✅ Hot-swapped IN: %s",
			name,
		)

	case "rmex":
		if len(args) < 1 {
			return "Usage: /rmex <exchange>"
		}

		name := strings.ToUpper(args[0])

		if err := a.connManager.RemoveConnector(name); err != nil {
			return fmt.Sprintf(
				"❌ Failed to remove %s: %v",
				name,
				err,
			)
		}

		return fmt.Sprintf(
			"🛑 Hot-swapped OUT: %s",
			name,
		)

	default:
		return "❓ Неизвестная команда. /help — список команд."
	}
}

package app

import (
	"context"
	"errors"
	"fmt"
	"log"
	"strings"
	"sync"
	"sync/atomic"
	"time"

	"crypto-screener/internal/domain"
)

type connectorEntry struct {
	cancel  context.CancelFunc
	wg      sync.WaitGroup
	stopped chan struct{}
}

func newConnectorEntry(
	ctx context.Context,
	name string,
	conn domain.ExchangeConnector,
	tickChan chan<- domain.MarketTick,
	fundingSink domain.FundingSink,
) *connectorEntry {
	if ctx == nil {
		ctx = context.Background()
	}

	entryCtx, cancel := context.WithCancel(ctx)

	e := &connectorEntry{
		cancel:  cancel,
		stopped: make(chan struct{}),
	}

	streams := []struct {
		suffix    string
		connectFn func(context.Context) error
	}{
		{
			"Spot",
			func(c context.Context) error {
				return conn.ConnectSpot(c, tickChan)
			},
		},
		{
			"Futures",
			func(c context.Context) error {
				return conn.ConnectFutures(c, tickChan)
			},
		},
		{
			"Funding",
			func(c context.Context) error {
				return conn.ConnectFunding(c, fundingSink)
			},
		},
	}

	e.wg.Add(len(streams))

	for _, s := range streams {
		s := s

		go runWithReconnect(
			entryCtx,
			&e.wg,
			fmt.Sprintf("%s/%s", name, s.suffix),
			s.connectFn,
		)
	}

	go func() {
		e.wg.Wait()
		close(e.stopped)
	}()

	return e
}

func (e *connectorEntry) stop(timeout time.Duration) error {
	e.cancel()

	timer := time.NewTimer(timeout)
	defer timer.Stop()

	select {
	case <-e.stopped:
		return nil

	case <-timer.C:
		return fmt.Errorf(
			"stop timed out after %v",
			timeout,
		)
	}
}

type ConnectorManager struct {
	mu sync.Mutex

	entries map[string]*connectorEntry

	// stopped изменяется под mu.
	stopped atomic.Bool

	// AddConnector, находящиеся между Phase 1 и Phase 4.
	activeAdds sync.WaitGroup

	// RemoveConnector, которые уже извлекли connector
	// из entries, но ещё не завершили stop().
	activeRemoves sync.WaitGroup

	// Горутинки остановки конфликтных connector'ов.
	backgroundWg sync.WaitGroup

	// Горутинки остановки connector'ов, созданных уже после
	// установки stopped=true.
	lateStopWg sync.WaitGroup

	// Делает StopAll() идемпотентным и заставляет concurrent callers
	// ждать завершения первого shutdown.
	stopOnce sync.Once
	stopDone chan struct{}

	stopTimeout time.Duration

	tickChan    chan<- domain.MarketTick
	fundingSink domain.FundingSink
}

type ConnectorManagerOption func(*ConnectorManager)

func WithStopTimeout(d time.Duration) ConnectorManagerOption {
	return func(cm *ConnectorManager) {
		if d > 0 {
			cm.stopTimeout = d
		}
	}
}

func NewConnectorManager(
	tickChan chan<- domain.MarketTick,
	fundingSink domain.FundingSink,
	opts ...ConnectorManagerOption,
) *ConnectorManager {
	cm := &ConnectorManager{
		entries:     make(map[string]*connectorEntry),
		stopTimeout: 15 * time.Second,
		stopDone:    make(chan struct{}),
		tickChan:    tickChan,
		fundingSink: fundingSink,
	}

	for _, opt := range opts {
		if opt != nil {
			opt(cm)
		}
	}

	return cm
}

var ErrManagerStopped = errors.New("connector manager is stopped")

func (cm *ConnectorManager) AddConnector(
	parentCtx context.Context,
	name string,
	conn domain.ExchangeConnector,
) error {
	name = strings.TrimSpace(name)

	if name == "" {
		return errors.New("connector name is empty")
	}

	if conn == nil {
		return errors.New("connector is nil")
	}

	// ------------------------------------------------------------
	// Phase 1
	// ------------------------------------------------------------
	cm.mu.Lock()

	if cm.stopped.Load() {
		cm.mu.Unlock()
		return ErrManagerStopped
	}

	cm.activeAdds.Add(1)

	old := cm.entries[name]
	delete(cm.entries, name)

	cm.mu.Unlock()

	defer cm.activeAdds.Done()

	// ------------------------------------------------------------
	// Phase 2
	// ------------------------------------------------------------
	if old != nil {
		if err := old.stop(cm.stopTimeout); err != nil {
			log.Printf(
				"⚠️ [%s] old connector stop warning: %v",
				name,
				err,
			)
		}
	}

	// ------------------------------------------------------------
	// Phase 3
	// ------------------------------------------------------------
	newEntry := newConnectorEntry(
		parentCtx,
		name,
		conn,
		cm.tickChan,
		cm.fundingSink,
	)

	// ------------------------------------------------------------
	// Phase 4
	// ------------------------------------------------------------
	cm.mu.Lock()
	defer cm.mu.Unlock()

	if cm.stopped.Load() {
		// StopAll уже был запущен.
		//
		// Add lateStopWg до activeAdds.Done().
		cm.lateStopWg.Add(1)

		go func() {
			defer cm.lateStopWg.Done()

			if err := newEntry.stop(cm.stopTimeout); err != nil {
				log.Printf(
					"⚠️ [%s] late stop warning: %v",
					name,
					err,
				)
			}
		}()

		return ErrManagerStopped
	}

	if _, conflict := cm.entries[name]; conflict {
		// Другой AddConnector успел зарегистрировать connector
		// с тем же именем.
		cm.backgroundWg.Add(1)

		go func() {
			defer cm.backgroundWg.Done()

			if err := newEntry.stop(cm.stopTimeout); err != nil {
				log.Printf(
					"⚠️ [%s] conflict stop warning: %v",
					name,
					err,
				)
			}
		}()

		return fmt.Errorf(
			"connector %q was concurrently replaced, retry if needed",
			name,
		)
	}

	cm.entries[name] = newEntry

	log.Printf(
		"✅ [%s] connector started",
		name,
	)

	return nil
}

func (cm *ConnectorManager) RemoveConnector(name string) error {
	name = strings.TrimSpace(name)

	if name == "" {
		return errors.New("connector name is empty")
	}

	// activeRemoves.Add() должен происходить под тем же mutex,
	// который используется StopAll().
	cm.mu.Lock()

	if cm.stopped.Load() {
		cm.mu.Unlock()
		return ErrManagerStopped
	}

	entry, exists := cm.entries[name]
	if !exists {
		cm.mu.Unlock()
		return fmt.Errorf(
			"connector %q not found",
			name,
		)
	}

	cm.activeRemoves.Add(1)

	delete(cm.entries, name)

	cm.mu.Unlock()

	defer cm.activeRemoves.Done()

	if err := entry.stop(cm.stopTimeout); err != nil {
		log.Printf(
			"⚠️ [%s] stop warning: %v",
			name,
			err,
		)

		return err
	}

	log.Printf(
		"🛑 [%s] connector stopped cleanly",
		name,
	)

	return nil
}

func (cm *ConnectorManager) StopAll() {
	cm.stopOnce.Do(func() {
		defer close(cm.stopDone)

		cm.mu.Lock()

		if cm.stopped.Load() {
			cm.mu.Unlock()
			return
		}

		cm.stopped.Store(true)

		entries := cm.entries
		cm.entries = make(map[string]*connectorEntry)

		cm.mu.Unlock()

		// --------------------------------------------------------
		// Step 1: registered connectors.
		// --------------------------------------------------------
		var registeredWg sync.WaitGroup

		for name, entry := range entries {
			name, entry := name, entry

			registeredWg.Add(1)

			go func() {
				defer registeredWg.Done()

				if err := entry.stop(cm.stopTimeout); err != nil {
					log.Printf(
						"⚠️ [%s] StopAll warning: %v",
						name,
						err,
					)
				} else {
					log.Printf(
						"🛑 [%s] stopped",
						name,
					)
				}
			}()
		}

		registeredWg.Wait()

		// --------------------------------------------------------
		// Step 2: conflict connectors.
		// --------------------------------------------------------
		cm.backgroundWg.Wait()

		// --------------------------------------------------------
		// Step 3:
		//
		// Ждём AddConnector, чтобы ни один lateStopWg.Add()
		// больше не мог произойти.
		// --------------------------------------------------------
		cm.activeAdds.Wait()

		// --------------------------------------------------------
		// Step 4:
		//
		// RemoveConnector мог извлечь connector из entries
		// непосредственно перед StopAll.
		// --------------------------------------------------------
		cm.activeRemoves.Wait()

		// --------------------------------------------------------
		// Step 5:
		//
		// Теперь все lateStopWg.Add() гарантированно произошли.
		// --------------------------------------------------------
		cm.lateStopWg.Wait()

		log.Println(
			"✅ ConnectorManager: all connectors stopped",
		)
	})

	// Если StopAll вызван конкурентно с другим StopAll,
	// второй caller ждёт завершения первого.
	<-cm.stopDone
}

func runWithReconnect(
	ctx context.Context,
	wg *sync.WaitGroup,
	streamName string,
	connectFn func(context.Context) error,
) {
	defer wg.Done()

	const (
		initialBackoff = 1 * time.Second
		maxBackoff     = 30 * time.Second
		resetThreshold = 10 * time.Second
	)

	backoff := initialBackoff

	for {
		if ctx.Err() != nil {
			return
		}

		start := time.Now()

		err := connectFn(ctx)

		duration := time.Since(start)

		if ctx.Err() != nil {
			return
		}

		if duration >= resetThreshold {
			backoff = initialBackoff
		}

		log.Printf(
			"⚠️ Stream [%s] disconnected after %v: %v. Retry in %v...",
			streamName,
			duration.Round(time.Millisecond),
			err,
			backoff,
		)

		timer := time.NewTimer(backoff)

		select {
		case <-ctx.Done():
			if !timer.Stop() {
				select {
				case <-timer.C:
				default:
				}
			}

			return

		case <-timer.C:
		}

		backoff = min(backoff*2, maxBackoff)
	}
}

package app

import (
	"errors"
	"sync"
	"sync/atomic"
	"time"

	"github.com/shopspring/decimal"
)

// Clock определяет интерфейс получения текущего времени для изоляции времени и тестирования.
type Clock interface {
	Now() time.Time
}

type realClock struct{}

func (realClock) Now() time.Time {
	return time.Now()
}

// ArbDirection указывает направление арбитражной сделки.
type ArbDirection int8

const (
	DirectionUnknown ArbDirection = iota
	SpotLongFuturesShort
	SpotShortFuturesLong
)

func (d ArbDirection) Valid() bool {
	return d == SpotLongFuturesShort || d == SpotShortFuturesLong
}

// ArbReason объясняет результат оценки арбитражной возможности.
type ArbReason string

const (
	ReasonProfitable               ArbReason = "PROFITABLE"
	ReasonBelowMinSpread           ArbReason = "BELOW_MIN_SPREAD"
	ReasonHazardWindow             ArbReason = "HAZARD_WINDOW"
	ReasonGracePeriod              ArbReason = "GRACE_PERIOD"
	ReasonWindowUnknown            ArbReason = "WINDOW_UNKNOWN"
	ReasonDataStale                ArbReason = "DATA_STALE"
	ReasonNoData                   ArbReason = "NO_DATA"
	ReasonStreamUnhealthy          ArbReason = "STREAM_UNHEALTHY"
	ReasonInvalidInput             ArbReason = "INVALID_INPUT"
	ReasonConfigurationUnavailable ArbReason = "CONFIGURATION_UNAVAILABLE"
)

// FundingConfig содержит конфигурационные параметры оценки фандинга.
type FundingConfig struct {
	MinSpread              decimal.Decimal
	MaxDataStaleness       time.Duration
	PreFundingHazardWindow time.Duration
	PostFundingGracePeriod time.Duration
}

// DefaultFundingConfig возвращает безопасную конфигурацию по умолчанию.
func DefaultFundingConfig() FundingConfig {
	return FundingConfig{
		MinSpread:              decimal.Zero,
		MaxDataStaleness:       15 * time.Second,
		PreFundingHazardWindow: 5 * time.Minute,
		PostFundingGracePeriod: 1 * time.Minute,
	}
}

func (c FundingConfig) normalize() FundingConfig {
	if c.MaxDataStaleness == 0 {
		c.MaxDataStaleness = 15 * time.Second
	}
	if c.PreFundingHazardWindow == 0 {
		c.PreFundingHazardWindow = 5 * time.Minute
	}
	if c.PostFundingGracePeriod == 0 {
		c.PostFundingGracePeriod = 1 * time.Minute
	}
	return c
}

func (c FundingConfig) Validate() error {
	if c.MinSpread.IsNegative() {
		return errors.New("MinSpread cannot be negative")
	}
	if c.MaxDataStaleness < 0 {
		return errors.New("MaxDataStaleness cannot be negative")
	}
	if c.PreFundingHazardWindow < 0 {
		return errors.New("PreFundingHazardWindow cannot be negative")
	}
	if c.PostFundingGracePeriod < 0 {
		return errors.New("PostFundingGracePeriod cannot be negative")
	}
	return nil
}

// FundingRecord представляет текущее состояние ставки фандинга по инструменту.
type FundingRecord struct {
	Symbol          string
	Rate            decimal.Decimal
	NextFundingTime time.Time
	LocalReceivedAt time.Time
	EventTime       time.Time // Биржевой таймстемп события (если поддерживается биржей)
}

func (r FundingRecord) IsStale(now time.Time, maxStaleness time.Duration) bool {
	if maxStaleness < 0 {
		return true
	}
	if maxStaleness == 0 {
		maxStaleness = 15 * time.Second
	}
	if r.LocalReceivedAt.IsZero() {
		return true
	}
	if now.Before(r.LocalReceivedAt) {
		// Защита от опережения времени: если расхождение больше 1 сек, считаем запись некорректной
		return r.LocalReceivedAt.Sub(now) > time.Second
	}
	return now.Sub(r.LocalReceivedAt) > maxStaleness
}

// ArbResult содержит результат оценки арбитражного окна.
type ArbResult struct {
	Profitable      bool
	Reason          ArbReason
	NetSpread       decimal.Decimal
	FundingRate     decimal.Decimal
	NextFundingTime time.Time
}

// FundingManager отвечает за хранение ставок и расчёт доходности с учётом временных окон.
type FundingManager struct {
	mu        sync.RWMutex
	rates     map[string]FundingRecord
	config    atomic.Pointer[FundingConfig]
	isHealthy atomic.Bool
	clock     Clock
}

// NewFundingManager создаёт новый экземпляр менеджера.
func NewFundingManager(cfg FundingConfig, clock Clock) (*FundingManager, error) {
	cfg = cfg.normalize()
	if err := cfg.Validate(); err != nil {
		return nil, err
	}
	if clock == nil {
		clock = realClock{}
	}

	fm := &FundingManager{
		rates: make(map[string]FundingRecord),
		clock: clock,
	}

	fm.config.Store(&cfg)
	fm.isHealthy.Store(true)

	return fm, nil
}

// UpdateConfig потокобезопасно обновляет конфигурацию с применением дефолтов и валидацией.
func (fm *FundingManager) UpdateConfig(cfg FundingConfig) error {
	cfg = cfg.normalize()
	if err := cfg.Validate(); err != nil {
		return err
	}
	fm.config.Store(&cfg)
	return nil
}

// SetStreamHealth меняет статус здоровья WebSocket-потока.
func (fm *FundingManager) SetStreamHealth(healthy bool) {
	fm.isHealthy.Store(healthy)
}

// UpdateFunding сохраняет актуальную ставку фандинга.
// Фильтрует устаревшие сообщения по EventTime (если задан с обеих сторон) или по LocalReceivedAt.
func (fm *FundingManager) UpdateFunding(
	symbol string,
	rate decimal.Decimal,
	nextFundingTime time.Time,
	eventTime time.Time,
) error {
	if symbol == "" {
		return errors.New("symbol must not be empty")
	}

	now := fm.clock.Now()

	fm.mu.Lock()
	defer fm.mu.Unlock()

	if existing, exists := fm.rates[symbol]; exists {
		switch {
		case !eventTime.IsZero() && !existing.EventTime.IsZero():
			if eventTime.Before(existing.EventTime) {
				return nil
			}
		default:
			if now.Before(existing.LocalReceivedAt) {
				return nil
			}
		}
	}

	fm.rates[symbol] = FundingRecord{
		Symbol:          symbol,
		Rate:            rate,
		NextFundingTime: nextFundingTime,
		LocalReceivedAt: now,
		EventTime:       eventTime,
	}

	return nil
}

// GetFunding возвращает снэпшот записи фандинга.
func (fm *FundingManager) GetFunding(symbol string) (FundingRecord, bool) {
	fm.mu.RLock()
	defer fm.mu.RUnlock()

	rec, exists := fm.rates[symbol]
	return rec, exists
}

// DeleteFunding удаляет запись по символу.
func (fm *FundingManager) DeleteFunding(symbol string) {
	fm.mu.Lock()
	defer fm.mu.Unlock()
	delete(fm.rates, symbol)
}

// EvictStale очищает устаревшие записи из памяти.
func (fm *FundingManager) EvictStale() {
	cfgPtr := fm.config.Load()
	if cfgPtr == nil {
		return
	}

	now := fm.clock.Now()
	cfg := *cfgPtr

	fm.mu.Lock()
	defer fm.mu.Unlock()

	for symbol, rec := range fm.rates {
		if rec.IsStale(now, cfg.MaxDataStaleness) {
			delete(fm.rates, symbol)
		}
	}
}

// CalcNetSpread рассчитывает чистый спрэд с учётом направления и ставки фандинга.
// Контракт единиц измерения: spread и rate передаются в виде десятичных дробей (0.01 = 1%, 0.001 = 0.1%).
func CalcNetSpread(spread, rate decimal.Decimal, direction ArbDirection) (decimal.Decimal, error) {
	switch direction {
	case SpotLongFuturesShort:
		// Длинная позиция спот / короткая фьючерс: получаем положительный фандинг
		return spread.Add(rate), nil
	case SpotShortFuturesLong:
		// Короткая позиция спот / длинная фьючерс: платим положительный фандинг
		return spread.Sub(rate), nil
	default:
		return decimal.Zero, errors.New("invalid arb direction")
	}
}

// EvaluateArb производит полную оценку арбитражного окна с учётом всех рисков.
func (fm *FundingManager) EvaluateArb(
	symbol string,
	spread decimal.Decimal,
	direction ArbDirection,
	now time.Time,
) ArbResult {
	if now.IsZero() {
		return ArbResult{Profitable: false, Reason: ReasonInvalidInput}
	}

	if !direction.Valid() {
		return ArbResult{Profitable: false, Reason: ReasonInvalidInput}
	}

	if !fm.isHealthy.Load() {
		return ArbResult{Profitable: false, Reason: ReasonStreamUnhealthy}
	}

	cfgPtr := fm.config.Load()
	if cfgPtr == nil {
		return ArbResult{Profitable: false, Reason: ReasonConfigurationUnavailable}
	}
	cfg := *cfgPtr

	rec, exists := fm.GetFunding(symbol)
	if !exists {
		return ArbResult{Profitable: false, Reason: ReasonNoData}
	}

	if rec.IsStale(now, cfg.MaxDataStaleness) {
		return ArbResult{
			Profitable:      false,
			Reason:          ReasonDataStale,
			FundingRate:     rec.Rate,
			NextFundingTime: rec.NextFundingTime,
		}
	}

	netSpread, err := CalcNetSpread(spread, rec.Rate, direction)
	if err != nil {
		return ArbResult{Profitable: false, Reason: ReasonInvalidInput}
	}

	reason := evaluateWindow(now, rec.NextFundingTime, cfg)
	if reason != ReasonProfitable {
		return ArbResult{
			Profitable:      false,
			Reason:          reason,
			NetSpread:       netSpread,
			FundingRate:     rec.Rate,
			NextFundingTime: rec.NextFundingTime,
		}
	}

	if netSpread.LessThan(cfg.MinSpread) {
		return ArbResult{
			Profitable:      false,
			Reason:          ReasonBelowMinSpread,
			NetSpread:       netSpread,
			FundingRate:     rec.Rate,
			NextFundingTime: rec.NextFundingTime,
		}
	}

	return ArbResult{
		Profitable:      true,
		Reason:          ReasonProfitable,
		NetSpread:       netSpread,
		FundingRate:     rec.Rate,
		NextFundingTime: rec.NextFundingTime,
	}
}

func evaluateWindow(now, nextFunding time.Time, cfg FundingConfig) ArbReason {
	if nextFunding.IsZero() {
		return ReasonWindowUnknown
	}

	// Окно 1: Время до фандинга
	if now.Before(nextFunding) {
		timeToFunding := nextFunding.Sub(now)
		if timeToFunding <= cfg.PreFundingHazardWindow {
			return ReasonHazardWindow
		}
		return ReasonProfitable
	}

	// Окно 2: Время после фандинга (Grace Period)
	timeSinceFunding := now.Sub(nextFunding)
	if timeSinceFunding <= cfg.PostFundingGracePeriod {
		return ReasonGracePeriod
	}

	return ReasonWindowUnknown
}

package app

import (
	"fmt"
	"time"

	"crypto-screener/internal/domain"

	"github.com/shopspring/decimal"

type NotificationRouter struct {
	userMgr     *domain.UserManager
	volProvider domain.VolumeProvider
	telegram    domain.TelegramSender
}

func NewNotificationRouter(userMgr *domain.UserManager, volProvider domain.VolumeProvider, tg domain.TelegramSender) *NotificationRouter {
	return &NotificationRouter{userMgr: userMgr, volProvider: volProvider, telegram: tg}
}

func (nr *NotificationRouter) ProcessSignal(signal *domain.ArbitrageSignal, isOpened bool) {
	users := nr.userMgr.GetAllUsers()
	if len(users) == 0 {
		return
	}

	// Группируем пользователей по таймфреймам, чтобы не считать объем для каждого отдельно
	tfGroups := make(map[domain.Timeframe][]*domain.User)
	for _, u := range users {
		tfGroups[u.Timeframe] = append(tfGroups[u.Timeframe], u)
	}

	// Считаем объемы для каждого уникального таймфрейма один раз
	tfVolumes := make(map[domain.Timeframe]decimal.Decimal)
	for tf := range tfGroups {
		tfVolumes[tf] = nr.volProvider.GetSymbolVolume(signal.Symbol, tf, signal.OpenedAt)
	}

	var targets []int64
	for tf, groupUsers := range tfGroups {
		vol := tfVolumes[tf]
		for _, u := range groupUsers {
			// Персональная фильтрация
			if signal.PeakSpread.GreaterThanOrEqual(u.MinSpread) && vol.GreaterThanOrEqual(u.MinVolume) {
				targets = append(targets, u.ChatID)
			}
		}
	}

	if len(targets) == 0 {
		return
	}

	var text string
	if isOpened {
		text = fmt.Sprintf("🚨 *SIGNAL OPENED*\nSymbol: `%s`\nType: %s\nExchanges: %s vs %s\nSpread: %s%%\nTime: %s",
			signal.Symbol, signal.SpreadType, signal.ExchangeA, signal.ExchangeB,
			signal.InitialSpread.Mul(decimal.NewFromInt(100)).StringFixed(2),
			signal.OpenedAt.Format(time.RFC3339))
	} else {
		text = fmt.Sprintf("✅ *SIGNAL CLOSED*\nSymbol: `%s`\nType: %s\nPeak: %s%%\nFinal: %s%%\nDuration: %s",
			signal.Symbol, signal.SpreadType,
			signal.PeakSpread.Mul(decimal.NewFromInt(100)).StringFixed(2),
			signal.FinalSpread.Mul(decimal.NewFromInt(100)).StringFixed(2),
			signal.Duration.Round(time.Second))
	}

	nr.telegram.Broadcast(text, targets)
}

package app

import (
	"context"
	"log"
	"sync"
	"time"

	"crypto-screener/internal/domain"

const persistenceTimeout = 5 * time.Second

type PersistenceWorker struct {
	dbChan     chan *domain.ArbitrageSignal
	signalRepo domain.SignalRepository
}

func NewPersistenceWorker(
	dbChan chan *domain.ArbitrageSignal,
	signalRepo domain.SignalRepository,
) *PersistenceWorker {
	return &PersistenceWorker{
		dbChan:     dbChan,
		signalRepo: signalRepo,
	}
}

func (w *PersistenceWorker) Start(
	ctx context.Context,
	wg *sync.WaitGroup,
) {
	defer wg.Done()

	log.Println("💾 Persistence worker started")

	// Не используем ctx.Done() для выхода.
	//
	// Application закрывает dbChan только после того,
	// как все producers завершили работу.
	//
	// Поэтому range гарантирует drain всей очереди.
	for signal := range w.dbChan {
		if signal == nil {
			log.Println("⚠️ Persistence worker received nil signal")
			continue
		}

		// Persistence должна завершить уже принятую работу,
		// даже если Application context был отменён.
		saveCtx, cancel := context.WithTimeout(
			context.WithoutCancel(ctx),
			persistenceTimeout,
		)

		err := w.signalRepo.SaveSignal(
			saveCtx,
			signal,
		)

		cancel()

		if err != nil {
			log.Printf(
				"❌ Failed to persist signal %s: %v",
				signal.ID,
				err,
			)

			continue
		}

		log.Printf(
			"💾 Signal persisted: %s",
			signal.ID,
		)
	}

	log.Println("💾 Persistence worker stopped")
}

package app

import (
	"hash/crc32"
	"sync"
	"time"

	"crypto-screener/internal/domain"

	"github.com/shopspring/decimal"
)

const numShards = 1024

type PriceState struct {
	Spot    decimal.Decimal
	Futures decimal.Decimal
}

type SymbolVolumeState struct {
	LastQuoteVol  decimal.Decimal
	Buckets       map[time.Time]decimal.Decimal
	LastPruneTime time.Time
}

type Shard struct {
	mu sync.Mutex

	prices  map[string]map[string]*PriceState
	volumes map[string]*SymbolVolumeState
}

type ShardedAggregator struct {
	shards [numShards]*Shard

	trackerChan chan<- domain.SpreadEvent
	funding     *FundingManager
	config      *domain.ScreenerConfig
}

func NewShardedAggregator(
	trackerChan chan<- domain.SpreadEvent,
	funding *FundingManager,
	cfg *domain.ScreenerConfig,
) *ShardedAggregator {
	sa := &ShardedAggregator{
		shards:      [numShards]*Shard{},
		trackerChan: trackerChan,
		funding:     funding,
		config:      cfg,
	}

	for i := 0; i < numShards; i++ {
		sa.shards[i] = &Shard{
			prices:  make(map[string]map[string]*PriceState),
			volumes: make(map[string]*SymbolVolumeState),
		}
	}

	return sa
}

func (sa *ShardedAggregator) ProcessTick(
	tick domain.MarketTick,
) {
	shardIdx := int(
		crc32.ChecksumIEEE(
			[]byte(tick.Symbol),
		) % numShards,
	)

	shard := sa.shards[shardIdx]

	// Проверяем конфигурацию до входа в shard lock.
	pairEnabled := sa.config.IsPairEnabled(tick.Symbol)

	shard.mu.Lock()

	if shard.prices[tick.Symbol] == nil {
		shard.prices[tick.Symbol] = make(map[string]*PriceState)
	}

	if shard.prices[tick.Symbol][tick.Exchange] == nil {
		shard.prices[tick.Symbol][tick.Exchange] = &PriceState{}
	}

	state := shard.prices[tick.Symbol][tick.Exchange]

	midPrice := tick.BestBid.
		Add(tick.BestAsk).
		Div(decimal.NewFromInt(2))

	switch tick.MarketType {
	case domain.MarketTypeSpot:
		state.Spot = midPrice

	case domain.MarketTypeFutures:
		state.Futures = midPrice

	default:
		shard.mu.Unlock()
		return
	}

	sa.updateVolumeState(
		shard,
		tick.Symbol,
		tick.QuoteVolume,
		tick.Timestamp,
	)

	if !pairEnabled {
		shard.mu.Unlock()
		return
	}

	// ------------------------------------------------------------
	// Считаем только данные, которые требуют доступа к shard state.
	// ------------------------------------------------------------

	var (
		crossValid bool
		minPrice   decimal.Decimal
		maxPrice   decimal.Decimal
		minEx      string
		maxEx      string

		intraValid bool
		spotPrice  decimal.Decimal
		futures    decimal.Decimal
	)

	// Cross-exchange.
	first := true

	for ex, exchangeState := range shard.prices[tick.Symbol] {
		if exchangeState.Futures.IsZero() {
			continue
		}

		if first {
			minPrice = exchangeState.Futures
			maxPrice = exchangeState.Futures
			minEx = ex
			maxEx = ex
			first = false
			continue
		}

		if exchangeState.Futures.LessThan(minPrice) {
			minPrice = exchangeState.Futures
			minEx = ex
		}

		if exchangeState.Futures.GreaterThan(maxPrice) {
			maxPrice = exchangeState.Futures
			maxEx = ex
		}
	}

	if !first && !minPrice.IsZero() {
		crossValid = true
	}

	// Intra-exchange.
	if !state.Spot.IsZero() && !state.Futures.IsZero() {
		intraValid = true
		spotPrice = state.Spot
		futures = state.Futures
	}

	// Важно: после этого момента никакого доступа к shard state.
	shard.mu.Unlock()

	// ------------------------------------------------------------
	// Все потенциально тяжёлые операции выполняются БЕЗ shard lock.
	// ------------------------------------------------------------

	if crossValid {
		spread := maxPrice.
			Sub(minPrice).
			Div(minPrice)

		if spread.GreaterThanOrEqual(
			sa.config.GetHardMinSpread(),
		) {
			if sa.funding.IsArbProfitable(
				tick.Symbol,
				spread,
				tick.Timestamp,
			) {
				sa.sendEventNonBlocking(
					domain.SpreadEvent{
						Symbol:      tick.Symbol,
						SpreadType:  domain.CrossExchange,
						Spread:      spread,
						ExchangeA:   minEx,
						ExchangeB:   maxEx,
						QuoteVolume: tick.QuoteVolume,
						Timestamp:   tick.Timestamp,
					},
				)
			}
		}
	}

	if intraValid {
		spread := futures.
			Sub(spotPrice).
			Abs().
			Div(spotPrice)

		if spread.GreaterThanOrEqual(
			sa.config.GetHardMinSpread(),
		) {
			if sa.funding.IsArbProfitable(
				tick.Symbol,
				spread,
				tick.Timestamp,
			) {
				sa.sendEventNonBlocking(
					domain.SpreadEvent{
						Symbol:      tick.Symbol,
						SpreadType:  domain.IntraExchange,
						Spread:      spread,
						ExchangeA:   tick.Exchange,
						ExchangeB:   tick.Exchange,
						QuoteVolume: tick.QuoteVolume,
						Timestamp:   tick.Timestamp,
					},
				)
			}
		}
	}
}

func (sa *ShardedAggregator) GetSymbolVolume(
	symbol string,
	tf domain.Timeframe,
	ts time.Time,
) decimal.Decimal {
	shardIdx := int(
		crc32.ChecksumIEEE(
			[]byte(symbol),
		) % numShards,
	)

	shard := sa.shards[shardIdx]

	shard.mu.Lock()
	defer shard.mu.Unlock()

	return sa.getSymbolVolumeInternal(
		shard,
		symbol,
		tf,
		ts,
	)
}

func (sa *ShardedAggregator) updateVolumeState(
	shard *Shard,
	symbol string,
	currentQVol decimal.Decimal,
	ts time.Time,
) {
	if shard.volumes[symbol] == nil {
		shard.volumes[symbol] = &SymbolVolumeState{
			Buckets: make(map[time.Time]decimal.Decimal),
		}
	}

	volState := shard.volumes[symbol]

	if !volState.LastQuoteVol.IsZero() {
		delta := currentQVol.Sub(
			volState.LastQuoteVol,
		)

		if delta.IsPositive() {
			minuteKey := ts.Truncate(time.Minute)

			volState.Buckets[minuteKey] =
				volState.Buckets[minuteKey].Add(delta)
		}
	}

	// Если биржа сбросила cumulative volume counter,
	// просто используем новое значение как baseline.
	volState.LastQuoteVol = currentQVol

	if ts.Sub(volState.LastPruneTime) > time.Minute {
		cutoff := ts.Add(-24 * time.Hour)

		for k := range volState.Buckets {
			if k.Before(cutoff) {
				delete(volState.Buckets, k)
			}
		}

		volState.LastPruneTime = ts
	}
}

func (sa *ShardedAggregator) getSymbolVolumeInternal(
	shard *Shard,
	symbol string,
	tf domain.Timeframe,
	ts time.Time,
) decimal.Decimal {
	volState, exists := shard.volumes[symbol]

	if !exists {
		return decimal.Zero
	}

	var duration time.Duration

	switch tf {
	case domain.TF_1m:
		duration = time.Minute

	case domain.TF_5m:
		duration = 5 * time.Minute

	case domain.TF_15m:
		duration = 15 * time.Minute

	case domain.TF_30m:
		duration = 30 * time.Minute

	case domain.TF_1h:
		duration = time.Hour

	case domain.TF_4h:
		duration = 4 * time.Hour

	default:
		duration = 24 * time.Hour
	}

	cutoff := ts.
		Truncate(time.Minute).
		Add(-duration)

	totalVol := decimal.Zero

	for k, v := range volState.Buckets {
		if k.After(cutoff) || k.Equal(cutoff) {
			totalVol = totalVol.Add(v)
		}
	}

	return totalVol
}

func (sa *ShardedAggregator) sendEventNonBlocking(
	event domain.SpreadEvent,
) {
	select {
	case sa.trackerChan <- event:
	default:
		// Backpressure policy:
		//
		// Market events are allowed to be dropped when trackerChan
		// is completely saturated. This prevents exchange ingestion
		// from blocking indefinitely.
		//
		// Если позже потребуется строгая гарантия доставки событий,
		// здесь должна появиться отдельная durable queue.
	}
}

package app

import (
	"fmt"
	"sync"
	"time"

	"crypto-screener/internal/domain"
)

type Tracker struct {
	mu sync.Mutex

	activeSignals map[string]*domain.ArbitrageSignal
	config        *domain.ScreenerConfig
	dbChan        chan<- *domain.ArbitrageSignal
	router        *NotificationRouter

	// Контролирует goroutines, отправляющие уведомления.
	notificationWg sync.WaitGroup
}

func NewTracker(
	cfg *domain.ScreenerConfig,
	dbChan chan<- *domain.ArbitrageSignal,
	router *NotificationRouter,
) *Tracker {
	return &Tracker{
		activeSignals: make(map[string]*domain.ArbitrageSignal),
		config:        cfg,
		dbChan:        dbChan,
		router:        router,
	}
}

func (t *Tracker) HandleEvent(
	event domain.SpreadEvent,
) {
	t.mu.Lock()

	exA := event.ExchangeA
	exB := event.ExchangeB

	if exA > exB {
		exA, exB = exB, exA
	}

	key := event.Symbol +
		":" +
		string(event.SpreadType) +
		":" +
		exA +
		":" +
		exB

	signal, exists := t.activeSignals[key]

	closeThreshold := t.config.GetCloseThreshold()

	var (
		notificationSignal *domain.ArbitrageSignal
		notificationOpen   bool
	)

	if exists {
		signal.UpdatePeak(event.Spread)

		if event.Spread.LessThanOrEqual(closeThreshold) {
			signal.Close(
				event.Timestamp,
				event.Spread,
			)

			delete(
				t.activeSignals,
				key,
			)

			// DB persistence.
			//
			// Здесь намеренно НЕ делаем default:
			// закрытый сигнал является уже сформированным
			// бизнес-событием и не должен молча теряться
			// только потому, что persistence queue заполнена.
			t.dbChan <- signal

			if t.router != nil {
				notificationSignal = signal
				notificationOpen = false
			}
		}
	} else {
		newSignal := domain.NewArbitrageSignal(
			event,
			event.Timestamp,
		)

		t.activeSignals[key] = newSignal

		if t.router != nil {
			notificationSignal = newSignal
			notificationOpen = true
		}
	}

	t.mu.Unlock()

	// Внешний side effect выполняется ПОСЛЕ освобождения Tracker mutex.
	if notificationSignal != nil {
		t.notify(
			notificationSignal,
			notificationOpen,
		)
	}
}

func (t *Tracker) notify(
	signal *domain.ArbitrageSignal,
	open bool,
) {
	if t.router == nil || signal == nil {
		return
	}

	t.notificationWg.Add(1)

	go func() {
		defer t.notificationWg.Done()

		// ProcessSignal — внешний side effect.
		t.router.ProcessSignal(
			signal,
			open,
		)
	}()
}

func (t *Tracker) Stop(
	timeout time.Duration,
) error {
	if timeout <= 0 {
		t.notificationWg.Wait()
		return nil
	}

	done := make(chan struct{})

	go func() {
		t.notificationWg.Wait()
		close(done)
	}()

	timer := time.NewTimer(timeout)
	defer timer.Stop()

	select {
	case <-done:
		return nil

	case <-timer.C:
		return fmt.Errorf(
			"tracker notification shutdown timed out after %v",
			timeout,
		)
	}
}
